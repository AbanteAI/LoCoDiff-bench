+        &self,
+        filter: Filter,
+        hw_measurement_acc: HwMeasurementAcc,
+        force: bool,
+    ) -> CollectionResult {

+
+        let Some(local_shard) = local_shard_guard.deref() else {

+
+        let mut next_offset = Some(ExtendedPointId::NumId(0));
